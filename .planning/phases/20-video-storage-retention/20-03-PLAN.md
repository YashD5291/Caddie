struct VideoInspection: Sendable, Equatable {
    let isPlayable: Bool
    let durationSeconds: Double
    let videoTrackCount: Int
    var isUsable: Bool   // isPlayable && durationSeconds > 0 && videoTrackCount > 0
}

enum VideoInspectionResult: Sendable, Equatable {
    case missing                    // nothing on disk, nothing to delete
    case inspected(VideoInspection)
    case unreadable(String)         // AVFoundation could not open it, e.g. "AVFoundationErrorDomain -11829: Cannot Open"
}

protocol VideoFileInspecting: Sendable {
    func inspect(_ url: URL) async -> VideoInspectionResult
}

struct AVAssetVideoInspector: VideoFileInspecting

final class MockVideoInspector: VideoFileInspecting, @unchecked Sendable   // Tests/Mocks only
    var defaultResult: VideoInspectionResult          // default .inspected(MockVideoInspector.usable)
    func setResult(_ result: VideoInspectionResult, for url: URL)   // keyed by url.lastPathComponent
    var inspectedURLs: [URL] { get }
    var inspectCallCount: Int { get }
    static let usable: VideoInspection                // isPlayable true, 30 s, 1 track
